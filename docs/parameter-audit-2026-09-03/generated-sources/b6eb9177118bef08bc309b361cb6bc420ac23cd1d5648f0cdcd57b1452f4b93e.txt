function capFor(p:Policy,s:RunState){
 if(p.cashoutAt&&s.floor>=p.cashoutAt)return 0;
 if(p.relay&&s.floor>=60&&s.restStops>0)return 0;
 if(p.relay&&s.floor<60)return 4;
 if(p.adaptive)return s.stress>=s.stressCap*.6||s.floor>=61?3:4;
 if(p.earlyRush)return s.floor<40?5:3;
 return p.cap;
}
function publicState(s:RunState):RunState{

}
function evaluator(p:Policy){
 const cache=new WeakMap<RunState,number>();
 return (state:RunState)=>{
  const cached=cache.get(state);if(cached!==undefined)return cached;
  const forecast=stressForecast(state),near=state.stress>state.stressCap*.55;
  let score=energySavings(state)*2;
  state.cabin.forEach((r,slot)=>{
   if(!r)return;
   const profile=riderProfile(r,state.cabin,slot),trip=Math.max(1,r.destination-state.floor);
   let fare=profile.hidden?Math.round(24*1.15):profile.fare;
   const paired=r.kind==='lover'&&hasNeighbour(state.cabin,slot,['lover']);
   const controlled=r.kind==='thief'&&hasNeighbour(state.cabin,slot,['cop','lawyer']);
   if(paired)fare*=2;if(controlled)fare+=5;
   if(r.kind==='ghost'&&hasNeighbour(state.cabin,slot,['exorcist']))fare+=6;
   if(r.kind==='coach')fare+=neighbours(slot).filter(i=>state.cabin[i]).length*3;
   if(r.kind!=='coach'&&hasNeighbour(state.cabin,slot,['coach']))fare=Math.ceil(fare*1.5);
   fare+=r.fareBonus+(bondStatus(r,state.cabin,slot).supported?cooperationBonus(state):0);
   const cost=trip*(near?2:1)+(r.kind==='child'&&!hasNeighbour(state.cabin,slot,['lover','musician','nurse'])?trip*.5:0);
   score+=fare/trip*1.5*(cost > r.patience ? .3 : 1);
   if(p.restAware&&cost<=r.patience)score+=Math.min(REST_STOP_CAP,REST_STOP_CAP-state.restStops)*Math.max(1,Math.floor(state.floor/30))*2/trip;
   if(paired)score+=1.5;
   if(r.kind==='thief')score+=controlled?1:3;
   if(r.kind==='drunk'&&hasNeighbour(state.cabin,slot,['musician','nurse']))score+=1;
   if(r.kind==='celebrity'&&neighbours(slot).filter(i=>state.cabin[i]).length===1)score+=3;
   if(p.favorite.includes(r.kind))score+=p.bias;
   if(p.rogue&&r.kind==='thief'&&!controlled)score+=3;
   if(p.quick)score+=2/trip;
   if(r.kind==='bomb'){
    const cops=neighbours(slot).flatMap(i=>state.cabin[i]?.kind==='cop'?[state.cabin[i]!]:[]);
    let fuse=r.fuse??0;
    for(let floor=state.floor+1;floor<=r.destination;floor++){
     if(!(floor%2===0&&cops.some(c=>c.destination>=floor&&c.patience>=floor-state.floor)))fuse--;
     // These policies do NOT intentionally use the impatience/fuse loophole.
     if(fuse<=0&&floor<r.destination){score-=1000;break;}
    }
   }
  });
  score-=(forecast.lowDelta+forecast.highDelta)/2*(near?p.risk*3:p.risk);
  if(state.stress+forecast.highDelta>=state.stressCap)score-=80;
  if(totalWeight(state.cabin)>state.weightCap)score-=20;
  if(p.lookahead){
   let future=publicState(state);const rng=seeded(34327+state.floor*13);
   for(let i=0;i<3&&future.status==='playing';i++)future=resolveFloor(future,rng);
   // Planner samples are independent of real future RNG and contain no future
   // offers. Mystery fare is masked even when a simulated arrival reveals it.
   score+=.3*((future.earned-state.earned)-(future.stress-state.stress)*p.risk+(future.energy-state.energy)*2-(future.status==='lost'?1000:0));
  }
  cache.set(state,score);return score;
 };
}
